# @farmfe/plugin-react
Support `React Jsx` and `React Refresh` for Farm.

## Installation

<Tabs>
  <TabItem value="npm" label="npm">
    <CodeBlock language="javascript" code={"npm install @farmfe/plugin-react react-refresh"} />
  </TabItem>
  <TabItem value="yarn" label="yarn">
    <CodeBlock language="javascript" code={"yarn add @farmfe/plugin-react react-refresh"} />
  </TabItem>
  <TabItem value="pnpm" label="pnpm">
    <CodeBlock language="javascript" code={"pnpm add @farmfe/plugin-react react-refresh"} />
  </TabItem>
</Tabs>

## Usage
`@farmfe/plugin-react` is a Rust plugin. Import its default export and call it inside the `plugins` array of `farm.config.ts`.
```ts {4}
import { UserConfig } from '@farmfe/core';
import react from '@farmfe/plugin-react';

const config: UserConfig = {
  plugins: [react({ /** options here */ })]
}
```

See [SWC Transform React Options](https://swc.rs/docs/configuration/compilation#jsctransformreact).
