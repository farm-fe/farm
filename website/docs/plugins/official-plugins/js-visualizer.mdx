# @farmfe/js-plugin-visualizer

Visualize Farm bundle size and module composition.

## Installation

<Tabs>
  <TabItem value="npm" label="npm">
    <CodeBlock language="javascript" code={"npm install @farmfe/js-plugin-visualizer"} />
  </TabItem>
  <TabItem value="yarn" label="yarn">
    <CodeBlock language="javascript" code={"yarn add @farmfe/js-plugin-visualizer"} />
  </TabItem>
  <TabItem value="pnpm" label="pnpm">
    <CodeBlock language="javascript" code={"pnpm add @farmfe/js-plugin-visualizer"} />
  </TabItem>
</Tabs>

## Usage

```ts {2,11}
import { defineConfig } from '@farmfe/core';
import visualizer from '@farmfe/js-plugin-visualizer';

export default defineConfig({
  compilation: {
    input: {
      index: './index.html'
    }
  },
  plugins: [
    visualizer()
  ]
});
```

## Options

```ts
export interface RecordViewerOptions {
  /**
   * Specify hostname.
   * @default 'localhost'
   */
  host?: string;

  /**
   * Specify port.
   * @default 9527
   */
  port?: number;
}
```

### `host`

Host for the visualizer server. Defaults to `localhost`.

### `port`

Port for the visualizer server. Defaults to `9527`.
