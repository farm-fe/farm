# @farmfe/plugin-virtual

Inspired By [@rollup/plugin-virtual](https://www.npmjs.com/package/@rollup/plugin-virtual)

A rust plugin for farm to easily use virtual module

## Installation

<Tabs>
  <TabItem value="npm" label="npm">
    <CodeBlock language="javascript" code={"npm install @farmfe/plugin-virtual"} />
  </TabItem>
  <TabItem value="yarn" label="yarn">
    <CodeBlock language="javascript" code={"yarn add @farmfe/plugin-virtual"} />
  </TabItem>
  <TabItem value="pnpm" label="pnpm">
    <CodeBlock language="javascript" code={"pnpm add @farmfe/plugin-virtual"} />
  </TabItem>
</Tabs>

## Usage

farm.config.ts

```typescript
import { defineConfig } from '@farmfe/core';

export default defineConfig({
  plugins: [
    [
      '@farmfe/plugin-virtual',
      {
        'virtual-module': 'export const a = 1',
        'src/01.js': 'export const module01 = "virtual-module"',
      },
    ],
  ],
});
```

index.js

```javascript
import { a } from 'virtual-module';
```

src/02.js

```javascript
import { module01 } from './01.js';
```
